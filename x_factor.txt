    FROM circle_posts p
    JOIN creators c ON p.creator_id = c.id
    WHERE p.positivity_score > 0.7
    ORDER BY p.posted_at DESC
    LIMIT ?
  `).bind(parseInt(limit)).all();
  
  // Cache for 15 mins
  await c.env.KV.put(`circle_feed:latest`, JSON.stringify(posts), { expirationTtl: 900 });
  
  return c.json(posts);
});

async function scorePost(content: string, env: Env): Promise<number> {
  const groq = new Groq({ apiKey: env.GROQ_API_KEY }); // or via Cloudflare AI Gateway
  
  const completion = await groq.chat.completions.create({
    model: "llama3-70b-8192", // or mixtral/fast model
    messages: [{
      role: "system",
      content: `You are a positivity curator for Kenyan creators. Score 0-1 for positivity, progress, unity. Support English/Swahili. Output only JSON: {"score": 0.85, "summary_sw": "...", "tags": ["tech", "success"]}`
    }, {
      role: "user",
      content: content
    }],
    temperature: 0.3,
    response_format: { type: "json_object" }
  });
  
  const result = JSON.parse(completion.choices[0].message.content!);
  return result.score;
}

// src/api/auth.ts
app.get('/api/auth/x', async (c) => {
  const state = crypto.randomUUID();
  const codeVerifier = crypto.randomUUID(); // PKCE

  await c.env.KV.put(`oauth_state:${state}`, codeVerifier, { expirationTtl: 600 });

  const authUrl = new URL('https://x.com/i/oauth2/authorize');
  authUrl.searchParams.append('response_type', 'code');
  authUrl.searchParams.append('client_id', c.env.X_CLIENT_ID);
  authUrl.searchParams.append('redirect_uri', `${new URL(c.req.url).origin}/api/auth/x/callback`);
  authUrl.searchParams.append('scope', 'tweet.read users.read offline.access');
  authUrl.searchParams.append('state', state);
  authUrl.searchParams.append('code_challenge', codeVerifier); // simple for demo; use proper SHA256 in prod
  authUrl.searchParams.append('code_challenge_method', 'plain');

  return c.redirect(authUrl.toString());
});

app.get('/api/auth/x/callback', async (c) => {
  const { code, state } = c.req.query();
  const verifier = await c.env.KV.get(`oauth_state:${state}`);

async function updateReciprocity(env: Env) {
  // Simple heuristic
  const creators = await env.DB.prepare("SELECT * FROM creators").all();

  for (const creator of creators.results) {
    // Count amplifications they made (track via self-reports or occasional API checks)
    const score = Math.min(100, creator.reciprocity_score + 5); // Base growth

    await env.DB.prepare("UPDATE creators SET reciprocity_score = ? WHERE id = ?")
